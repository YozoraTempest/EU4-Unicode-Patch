cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" PATCH_VERSION)
string(STRIP "${PATCH_VERSION}" PATCH_VERSION)
if(NOT PATCH_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "VERSION must contain a three-part numeric version")
endif()
project(EU4UnicodePatch VERSION ${PATCH_VERSION} LANGUAGES C CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
if(DEFINED SHOWINCLUDES_PREFIX_OVERRIDE)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${SHOWINCLUDES_PREFIX_OVERRIDE}")
    set(CMAKE_C_CL_SHOWINCLUDES_PREFIX "${SHOWINCLUDES_PREFIX_OVERRIDE}")
    set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "${SHOWINCLUDES_PREFIX_OVERRIDE}")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
    include(cmake/windows.cmake)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    include(cmake/linux-core.cmake)
else()
    message(FATAL_ERROR "EU4 Unicode Patch supports Windows and Linux build targets")
endif()
