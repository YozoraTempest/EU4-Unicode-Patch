cmake_minimum_required(VERSION 3.24)
project(EU4UnicodePatch LANGUAGES C CXX ASM_MASM)
if(DEFINED SHOWINCLUDES_PREFIX_OVERRIDE)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${SHOWINCLUDES_PREFIX_OVERRIDE}")
    set(CMAKE_C_CL_SHOWINCLUDES_PREFIX "${SHOWINCLUDES_PREFIX_OVERRIDE}")
    set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "${SHOWINCLUDES_PREFIX_OVERRIDE}")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(unicode_text STATIC src/unicode_text.cpp)
target_include_directories(unicode_text PUBLIC include PRIVATE vendor/utfcpp/source)
add_library(unicode_services STATIC src/unicode_services.cpp)
target_link_libraries(unicode_services PUBLIC unicode_text icu)
add_library(unicode_layout STATIC src/unicode_layout.cpp)
target_link_libraries(unicode_layout PUBLIC unicode_services dwrite d2d1 windowscodecs ole32)
target_compile_definitions(unicode_layout PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_subdirectory(vendor/minhook)
add_library(eu4_unicode_probe SHARED src/plugin.cpp src/text_hooks.asm src/map_hooks.asm)
target_include_directories(eu4_unicode_probe PRIVATE src)
target_link_libraries(eu4_unicode_probe PRIVATE unicode_services minhook bcrypt)
target_compile_definitions(eu4_unicode_probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(unicode_text_tests tests/unicode_text_tests.cpp)
add_executable(guard_host tests/guard_host.cpp)
target_link_libraries(unicode_text_tests PRIVATE unicode_text)
add_executable(unicode_services_tests tests/unicode_services_tests.cpp)
target_link_libraries(unicode_services_tests PRIVATE unicode_services)
add_executable(unicode_layout_tests tests/unicode_layout_tests.cpp)
target_link_libraries(unicode_layout_tests PRIVATE unicode_layout)
foreach(target unicode_text unicode_services unicode_layout eu4_unicode_probe unicode_text_tests unicode_services_tests unicode_layout_tests)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /utf-8 /EHsc>)
endforeach()
enable_testing()
add_test(NAME unicode_text COMMAND unicode_text_tests)
add_test(NAME unicode_services COMMAND unicode_services_tests)
add_test(NAME unicode_layout COMMAND unicode_layout_tests)
