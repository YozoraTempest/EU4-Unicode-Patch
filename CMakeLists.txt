cmake_minimum_required(VERSION 3.24)
project(EU4UnicodePatch LANGUAGES C CXX ASM_MASM)
if(DEFINED SHOWINCLUDES_PREFIX_OVERRIDE)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${SHOWINCLUDES_PREFIX_OVERRIDE}")
    set(CMAKE_C_CL_SHOWINCLUDES_PREFIX "${SHOWINCLUDES_PREFIX_OVERRIDE}")
    set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "${SHOWINCLUDES_PREFIX_OVERRIDE}")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(unicode_text STATIC src/unicode_text.cpp src/native_text_event.cpp)
target_include_directories(unicode_text PUBLIC include PRIVATE vendor/utfcpp/source)
add_library(unicode_services STATIC src/unicode_services.cpp)
target_link_libraries(unicode_services PUBLIC unicode_text icu)
add_library(unicode_editor STATIC src/unicode_editor.cpp)
target_link_libraries(unicode_editor PUBLIC unicode_services)
add_library(unicode_search STATIC src/unicode_search.cpp)
target_link_libraries(unicode_search PUBLIC unicode_services)
add_library(unicode_layout STATIC src/unicode_layout.cpp)
target_link_libraries(unicode_layout PUBLIC unicode_services dwrite d2d1 windowscodecs ole32)
target_compile_definitions(unicode_layout PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(fontpack tools/fontpack.cpp)
target_link_libraries(fontpack PRIVATE unicode_layout)
add_subdirectory(vendor/minhook)
add_library(glyph_registry STATIC src/glyph_registry.cpp)
target_include_directories(glyph_registry PUBLIC include)
add_library(eu4_unicode_probe SHARED src/plugin.cpp src/text_hooks.asm src/map_hooks.asm src/font_hooks.asm src/path_hooks.asm)
set_source_files_properties(src/text_hooks.asm src/map_hooks.asm src/font_hooks.asm
    PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/hook_context.inc")
target_include_directories(eu4_unicode_probe PRIVATE src)
target_link_options(eu4_unicode_probe PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/eu4_unicode_probe.map")
target_link_libraries(eu4_unicode_probe PRIVATE unicode_editor unicode_search glyph_registry minhook bcrypt)
target_compile_definitions(eu4_unicode_probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(unicode_text_tests tests/unicode_text_tests.cpp)
add_executable(guard_host tests/guard_host.cpp)
target_link_libraries(unicode_text_tests PRIVATE unicode_text)
add_executable(unicode_services_tests tests/unicode_services_tests.cpp)
target_link_libraries(unicode_services_tests PRIVATE unicode_services)
add_executable(unicode_layout_tests tests/unicode_layout_tests.cpp)
target_link_libraries(unicode_layout_tests PRIVATE unicode_layout)
add_executable(glyph_registry_tests tests/glyph_registry_tests.cpp)
target_link_libraries(glyph_registry_tests PRIVATE glyph_registry)
add_executable(unicode_editor_tests tests/unicode_editor_tests.cpp)
target_link_libraries(unicode_editor_tests PRIVATE unicode_editor)
add_executable(unicode_search_tests tests/unicode_search_tests.cpp)
target_link_libraries(unicode_search_tests PRIVATE unicode_search)
foreach(target unicode_text unicode_services unicode_editor unicode_search unicode_layout glyph_registry fontpack eu4_unicode_probe unicode_text_tests unicode_services_tests unicode_layout_tests glyph_registry_tests unicode_editor_tests unicode_search_tests)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /utf-8 /EHsc>)
endforeach()
enable_testing()
add_test(NAME unicode_text COMMAND unicode_text_tests)
add_test(NAME unicode_services COMMAND unicode_services_tests)
add_test(NAME unicode_layout COMMAND unicode_layout_tests)
add_test(NAME glyph_registry COMMAND glyph_registry_tests)
add_test(NAME unicode_editor COMMAND unicode_editor_tests)
add_test(NAME unicode_search COMMAND unicode_search_tests)
