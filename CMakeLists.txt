cmake_minimum_required(VERSION 3.24)
project(EU4UnicodePatch LANGUAGES C CXX ASM_MASM)
if(DEFINED SHOWINCLUDES_PREFIX_OVERRIDE)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${SHOWINCLUDES_PREFIX_OVERRIDE}")
    set(CMAKE_C_CL_SHOWINCLUDES_PREFIX "${SHOWINCLUDES_PREFIX_OVERRIDE}")
    set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "${SHOWINCLUDES_PREFIX_OVERRIDE}")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(unicode_text STATIC src/unicode_text.cpp src/native_text_event.cpp)
target_include_directories(unicode_text PUBLIC include PRIVATE vendor/utfcpp/source)
add_library(unicode_services STATIC src/unicode_services.cpp)
target_link_libraries(unicode_services PUBLIC unicode_text icu)
add_library(unicode_editor STATIC src/unicode_editor.cpp)
target_link_libraries(unicode_editor PUBLIC unicode_services)
add_library(unicode_search STATIC src/unicode_search.cpp)
target_link_libraries(unicode_search PUBLIC unicode_services)
add_library(unicode_layout STATIC src/unicode_layout.cpp)
target_link_libraries(unicode_layout PUBLIC unicode_services dwrite d2d1 windowscodecs ole32)
target_compile_definitions(unicode_layout PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(font_atlas_assets STATIC src/font_atlas_assets.cpp)
target_link_libraries(font_atlas_assets PUBLIC unicode_layout)
add_executable(fontpack tools/fontpack.cpp)
target_link_libraries(fontpack PRIVATE font_atlas_assets)
add_subdirectory(vendor/minhook)
add_library(glyph_registry STATIC src/glyph_registry.cpp)
target_include_directories(glyph_registry PUBLIC include)
add_library(scalar_glyph STATIC src/scalar_glyph.cpp)
target_link_libraries(scalar_glyph PUBLIC unicode_layout glyph_registry)
add_library(font_draw_batches STATIC src/font_draw_batches.cpp)
target_include_directories(font_draw_batches PUBLIC include)
add_library(native_font_atlas STATIC src/native_font_atlas.cpp src/native_font_draw.cpp)
target_link_libraries(native_font_atlas PUBLIC scalar_glyph font_draw_batches PRIVATE minhook d3d9)
target_compile_definitions(native_font_atlas PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(native_ime STATIC src/native_ime.cpp)
target_include_directories(native_ime PUBLIC include)
target_link_libraries(native_ime PUBLIC imm32)
target_compile_definitions(native_ime PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(font_assets STATIC src/font_assets.cpp)
target_include_directories(font_assets PUBLIC include)
add_library(eu4_unicode_probe SHARED src/plugin.cpp src/text_hooks.asm src/map_hooks.asm src/font_hooks.asm src/path_hooks.asm)
add_library(eu4_unicode_patch SHARED src/plugin.cpp src/text_hooks.asm src/map_hooks.asm src/font_hooks.asm src/path_hooks.asm)
target_compile_definitions(eu4_unicode_probe PRIVATE EU4_UNICODE_RESEARCH)
set_source_files_properties(src/text_hooks.asm src/map_hooks.asm src/font_hooks.asm
    PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/hook_context.inc")
foreach(target eu4_unicode_probe eu4_unicode_patch)
    target_include_directories(${target} PRIVATE src)
    target_link_options(${target} PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/${target}.map")
    target_link_libraries(${target} PRIVATE unicode_editor unicode_search glyph_registry native_ime native_font_atlas font_assets font_atlas_assets minhook bcrypt)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
endforeach()
add_library(version_proxy SHARED src/version_proxy.cpp src/version_proxy.asm src/version_proxy.def)
set_target_properties(version_proxy PROPERTIES OUTPUT_NAME VERSION)
target_compile_definitions(version_proxy PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(font_assets_tests tests/font_assets_tests.cpp)
add_executable(font_draw_batches_tests tests/font_draw_batches_tests.cpp)
target_link_libraries(font_draw_batches_tests PRIVATE font_draw_batches)
target_link_libraries(font_assets_tests PRIVATE font_assets)
add_executable(font_cache_tests tests/font_cache_tests.cpp)
target_link_libraries(font_cache_tests PRIVATE font_atlas_assets scalar_glyph)
add_executable(version_proxy_tests tests/version_proxy_tests.cpp)
target_link_libraries(version_proxy_tests PRIVATE version)
add_library(loader_fixture SHARED tests/loader_fixture.cpp)
add_executable(unicode_text_tests tests/unicode_text_tests.cpp)
add_executable(guard_host tests/guard_host.cpp)
target_link_libraries(unicode_text_tests PRIVATE unicode_text)
add_executable(unicode_services_tests tests/unicode_services_tests.cpp)
target_link_libraries(unicode_services_tests PRIVATE unicode_services)
add_executable(unicode_layout_tests tests/unicode_layout_tests.cpp)
target_link_libraries(unicode_layout_tests PRIVATE unicode_layout)
add_executable(glyph_registry_tests tests/glyph_registry_tests.cpp)
target_link_libraries(glyph_registry_tests PRIVATE glyph_registry)
add_executable(unicode_editor_tests tests/unicode_editor_tests.cpp)
target_link_libraries(unicode_editor_tests PRIVATE unicode_editor)
add_executable(unicode_search_tests tests/unicode_search_tests.cpp)
target_link_libraries(unicode_search_tests PRIVATE unicode_search)
add_executable(native_ime_tests tests/native_ime_tests.cpp)
target_link_libraries(native_ime_tests PRIVATE native_ime)
add_executable(open_font_tests tests/open_font_tests.cpp)
target_link_libraries(open_font_tests PRIVATE scalar_glyph)
add_executable(scalar_probe tools/scalar-probe.cpp)
target_link_libraries(scalar_probe PRIVATE scalar_glyph)
add_executable(native_font_atlas_tests tests/native_font_atlas_tests.cpp)
target_link_libraries(native_font_atlas_tests PRIVATE native_font_atlas minhook d3d9)
target_compile_definitions(native_font_atlas_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(native_font_pages_tests tests/native_font_pages_tests.cpp)
target_link_libraries(native_font_pages_tests PRIVATE native_font_atlas minhook d3d9)
target_compile_definitions(native_font_pages_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(native_font_pages_tests PRIVATE /W4 /utf-8 /EHsc)
foreach(target unicode_text unicode_services unicode_editor unicode_search unicode_layout glyph_registry scalar_glyph font_draw_batches native_font_atlas native_ime font_assets font_atlas_assets fontpack eu4_unicode_probe eu4_unicode_patch version_proxy font_assets_tests font_draw_batches_tests font_cache_tests version_proxy_tests unicode_text_tests unicode_services_tests unicode_layout_tests glyph_registry_tests unicode_editor_tests unicode_search_tests native_ime_tests open_font_tests scalar_probe native_font_atlas_tests)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /utf-8 /EHsc>)
endforeach()
enable_testing()
add_test(NAME unicode_text COMMAND unicode_text_tests)
add_test(NAME unicode_services COMMAND unicode_services_tests)
add_test(NAME unicode_layout COMMAND unicode_layout_tests)
add_test(NAME glyph_registry COMMAND glyph_registry_tests)
add_test(NAME unicode_editor COMMAND unicode_editor_tests)
add_test(NAME unicode_search COMMAND unicode_search_tests)
add_test(NAME native_ime COMMAND native_ime_tests)
add_test(NAME font_assets COMMAND font_assets_tests)
add_test(NAME font_draw_batches COMMAND font_draw_batches_tests)
add_test(NAME font_cache COMMAND font_cache_tests "${CMAKE_CURRENT_BINARY_DIR}/runtime-font-assets")
add_test(NAME version_proxy COMMAND version_proxy_tests $<TARGET_FILE:version_proxy>)
